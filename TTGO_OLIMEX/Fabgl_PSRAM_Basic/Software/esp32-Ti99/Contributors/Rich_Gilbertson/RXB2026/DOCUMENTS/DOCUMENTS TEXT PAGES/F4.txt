          10000 SUB CONFIRMF18
          10010 CALL F18UNLOCK
          10020 CALL POKEV(4096,0,0,7,32,16,0,3,64)
          10030 CALL POKER(54,16;55,2)
          10040 CALL PEEKV(4096,X)
          10050 IF X<>0 THEN 10080
          10060 PRINT "F18A REQUIRED!"
          10070 END
          10080 SUBEND

          12000 SUB LOADPALETTE(PALNUM)
          12010 ! VPOKE YOUR 32-BYTES (16 COLORS) 
                AT VDP 4116 BEFORE CALLING!
          12020 ! WILL OVERWRITE THE PROGRAM AT VDP 4096
          12030 ! PALNUM IS 0-3 FOR WHICH PALETTE TO LOAD
          12040 ! IS VDP@>1000 ALWAYS A SAFE LOCATION?
                MAYBE DEPENDS ON PROGRAM
          12050 ! THIS IS THE PROGRAM LOOP BELOW
          12060 IF PALNUM>=0 AND PALNUM<=3 THEN 12080
          12070 BREAK::! ILLEGAL PALETTE NUMBER 
                - BREAK INSTEAD OF CORRUPT VDP RAM
          12080 CALL POKEV(4096,2,0,16,20,2,1,80,PALNUM*32,2,2,0,16)
          12090 CALL POKEV(4108,204,112,6,2,22,253,3,64)
          12100 ! ASSUMES THE CALLER ALREADY LOADED THE PALETTE
          12110 CALL POKER(54,16;55,0)
          12120 SUBEND

          13000 SUB RESETF18A
          13010 ! FIRST RESTORE THE DEFAULT TI PALETTE
          13020 ! SINCE THIS CAN'T BE DONE AUTOMATICALLY!
          13030 ! IF WE NEVER CHANGED PALETTE 0 
                (IE: TRY CHANGING THE CODE ABOVE TO CHANGE PALETTES 
                1 AND 2 INSTEAD OF 0 AND 1.
          13040 ! USE REGISTER 24 TO CHANGE THE TEXT LAYER TO THE NEW PALETTE)
          13050 ! ANYWAY, IF WE DIDN'T CHANGE PALETTE 0, 
                THEN THE RESET WOULD BE ENOUGH!
          13060 ! JUST POKE IN THE VALUES FOR THE DEFAULT PALETTE, 
                AND THEN LOAD PALETTE 0
          13070 CALL POKEV(4116,0,0,0,0,2,195,5,214)
          13080 CALL POKEV(4124,5,79,7,111,13,84,4,239)
          13090 CALL POKEV(4132,15,84,15,118,13,195,14,214)
          13100 CALL POKEV(4140,2,178,12,92,12,204,15,255)
          13110 CALL LOADPALETTE(0)
          13120 ! THEN JUST POKE THE RESET REGISTER AND EVERYTHING 
                ELSE IS RESTORED
          13130 CALL POKER(50,128)
          13140 SUBEND
